# Regression corpus for the `hardcoded-secret` / `known-secret-format` rules in
# scripts/audit-sast.ts. Read and asserted by the auditor ON EVERY RUN — if any MUST-CATCH
# line stops being caught, or any MUST-IGNORE line starts being flagged, the auditor fails
# before it scans anything.
#
# WHY THIS EXISTS
#
# The previous `hardcoded-secret` rule sat in the compliance chain, was reported green, and
# could not see a single real credential. It was measured against the passwords found live in
# the DemoCRM webroot in July 2026 — served to the open internet — and missed every one, for
# six independent reasons (documented in audit-sast.ts). Nobody noticed, because a rule with
# no test only tells you it found nothing; it never tells you it CANNOT find anything.
#
# The values below are SYNTHETIC. They are not, and never were, real credentials — they carry
# the SHAPES that defeated the old rule (a 14-character alphanumeric password; a password
# ending in punctuation; camelCase and snake_case names; an XML attribute pair), which is what
# the regression is actually about. Do not replace them with real ones: a dead credential in a
# fixture is still a credential in source, and these rules would rightly flag it.
#
# Format:  CATCH <line>   — the rule MUST report this
#          IGNORE <line>  — the rule MUST NOT report this
# Blank lines and #-comments are skipped.

# --- shapes the old rule was blind to -----------------------------------------------------
# 14 chars, alphanumeric: under the old 20-char minimum
CATCH $prodPassword = 'q7Xm2Kd9Lp4Rt1';
# member access: '->' was not an accepted boundary
CATCH $this->crm_password = 'q7Xm2Kd9Lp4Rt1';
# quoted array key, PHP fat arrow
CATCH $cfg = ['password' => 'q7Xm2Kd9Lp4Rt1'];
# trailing '//' comment: the old rule skipped ANY line containing '//'
CATCH $withNote = ['password' => 'q7Xm2Kd9Lp4Rt1']; // prod
# punctuation in the value: '!' was outside the old character class
CATCH $testPass = 'SynthDB_2026_Secret!';
# XML/env attribute pair — .xml was not even scanned
CATCH <env name="TEST_DB_PASS" value="SynthDB_2026_Secret!"/>
# camelCase and SCREAMING_CASE names
CATCH const apiKey = 'a9F2kQ7xZ1mB4nR8'
CATCH DATABASE_PASSWORD="hX3pW9qL2vT6yC1s"

# --- provider token formats, matched on shape alone ---------------------------------------
CATCH const k = 'AKIA1234567890ABCDEF'
CATCH const g = 'ghp_abcdefghijklmnopqrstuvwxyz0123456789'
CATCH const a = 'sk-ant-api03-abcdefghijklmnopqrstuvwxyz'

# --- must stay quiet: flagging these is how an auditor loses its authority -----------------
# UI labels
IGNORE $label_password = "Password";
IGNORE const ariaLabelPassword = "Enter password"
# translations — prose in any alphabet is not a secret
IGNORE $lang = ['password' => 'Contraseña'];
IGNORE $lang = ['new_password' => 'New Password'];
# templates: committing these is the CORRECT behaviour
IGNORE $tpl = ['password' => 'CHANGE_ME'];
IGNORE $tpl = ['password' => '<YOUR_PASSWORD>'];
# throwaway app fixture logins, not credentials
IGNORE $fixture = ['username' => 'testadmin', 'password' => 'test_pass'];
# *_FILE names a LOCATION, not a secret — this is the sops-nix convention and the whole point
IGNORE "ENGRAM_COGNEE_PASSWORD_FILE": "/run/secrets/engram-cognee-password"
IGNORE DB_PASSWORD_FILE=/run/secrets/db-password

# --- shell expansions are not literals (added 2026-08-10) ---------------------------------
# A per-run random value: the assignment is exactly the shape of a hardcoded secret, and the
# value is generated at runtime. Flagging it taught authors to reword rather than to stop
# hardcoding, which is how an auditor loses its authority.
IGNORE LICENSE_HMAC_SECRET="$(head -c 32 /dev/urandom | base64 | tr -d '\n')"
IGNORE DB_PASSWORD="$(vault read -field=pw secret/db)"
IGNORE API_TOKEN=`cat /run/secrets/token`
# Passing an already-resolved variable through to a child process.
IGNORE LICENSE_HMAC_SECRET="$LICENSE_HMAC_SECRET"
IGNORE DB_PASSWORD=${DB_PASSWORD}
# ...but a literal that merely CONTAINS a dollar sign is still a literal, and still a leak.
CATCH DB_PASSWORD="pa$$w0rdLiteral99"
CATCH $cfg = ['password' => 'Str0ng$Pass!2026'];

# --- a constant holding the NAME of an env var is a POINTER, not a credential (2026-08-27) ---
# Same class as PASSWORD_FILE=/run/secrets/… above: this is what a hardcoded credential is
# replaced BY, so flagging it tells an author that doing the right thing is a violation.
IGNORE hardcoded-secret const TOKEN_ENV = 'APP_GITHUB_TOKEN'
IGNORE hardcoded-secret const SECRET_VAR_NAME = "APP_SESSION_SECRET"
# The near-miss in the other direction MUST still be caught: the exemption is on the value being
# an identifier, so a real token assigned to the same-shaped constant is not exempt.
CATCH hardcoded-secret const TOKEN_ENV = 'ghp_0123456789abcdefghijklmnop'
CATCH hardcoded-secret const PASSWORD_ENV = 'hunter2-real-password'

# --- a camelCase location name is a location too (added 2026-09-10) -----------------------
# The *_URL / *_FILE exemption read snake_case only, so Qepton's `accessTokenUrl` — GitHub's
# public OAuth token ENDPOINT — was reported as a hardcoded secret. Same class as
# PASSWORD_FILE above, in the spelling the rule's own comments warn it used to miss.
IGNORE const accessTokenUrl = 'https://github.com/login/oauth/access_token'
IGNORE const apiKeyLocation = "https://vault.example.test/v1/kv/app"
# `…Profile` is not `…File`: the camelCase suffix is case-sensitive, so this stays a finding.
CATCH const tokenProfile = 'q7Xm2Kd9Lp4Rt1'
# ...and a URL that CARRIES the credential is still a leak, under either spelling.
CATCH const tokenUrl = 'https://svc:q7Xm2Kd9Lp4Rt1@auth.example.test/token'
CATCH SECRET_URL="https://hooks.example.test/in?token=q7Xm2Kd9Lp4Rt1"

# --- a path rooted at a variable is a path (added 2026-09-18) ------------------------------
# Gantry's sops wrapper names the file it decrypts from, and the rule reported it as a hardcoded
# secret. It already exempts a value starting with / . or ~ as a path; "$ROOT/…" is the same thing
# built from a variable, and in a shell or PHP double-quoted string it is an expansion besides.
IGNORE SECRETS="$ROOT/secrets/app.yaml"
IGNORE API_TOKEN="$XDG_RUNTIME_DIR/app-token"
# The exemption is a variable at the START, followed by a separator. A literal that begins with a
# dollar-word and no separator, or carries "$X/" later on, is still a literal.
CATCH DB_PASSWORD="$uper-Secret.99x"
CATCH DB_PASSWORD="Xy7$ab/Cd99efgh"

# --- a DECLARATION whose key merely contains "token" ---------------------------------------
# Measured 2026-09-20: `"audit:token-reach": "…"` in scripts/data/auditor-contracts.json. The key
# capture starts after the namespace colon, so the rule read `token-reach` assigned a paragraph of
# English. Prose is not a credential, and the fix belongs in the rule: renaming the auditor to
# please the regex is the input-rewording never-game-auditors.md refuses.
IGNORE   "audit:token-reach": "docs/SECRETS.md § GitHub tokens — run at mint, at rotation, and when bringing a repository live"
IGNORE   const tokenHelp = 'pass the token as a file, never on the command line'
# …and the gate that keeps the exemption narrow: four groups is not four WORDS once digits appear.
CATCH apiKey = '3f9a b2c1 77de 9911'
CATCH token: 'Qx7 Lm2 Rt9 Zp4 Kd1'

# --- an ensure_ascii JSON escape is an encoded CHARACTER, not entropy -----------------------
# The same declaration line, as the file actually stores it: `\u00a7` for § and `\u2014` for —.
# Four hex digits per escape were enough to make a paragraph of English look like a credential.
IGNORE   "audit:token-reach": "docs/SECRETS.md \u00a7 GitHub tokens \u2014 run at mint, at rotation, and when bringing a repository live"
# The CATCH half of that normalisation: a real secret is not rescued by containing an escape.
CATCH password = 'a\u0041b9Xq2Lm7'

# --- UNQUOTED values in config files (added 2026-10-01) ------------------------------------
# Every rule above read a QUOTED value only. YAML, .env, shell and INI are written unquoted as
# often as not, so `POSTGRES_PASSWORD: <literal>` in a committed docker-compose.yml passed the
# scan clean in the template and in Gantry, in the exact file class the root-level scan names.
CATCH POSTGRES_PASSWORD: q7Xm2Kd9Lp4Rt1
CATCH       - POSTGRES_PASSWORD=q7Xm2Kd9Lp4Rt1
CATCH export API_TOKEN=q7Xm2Kd9Lp4Rt1
CATCH db_password = q7Xm2Kd9Lp4Rt1   # prod
# A shell env prefix on a command, and one line of a continued block: no whole-line shape fits.
CATCH PGPASSWORD=q7Xm2Kd9Lp4Rt1 psql -h db -U app
CATCH   JWT_SECRET=e2e-fixed-jwt-2026-secret \
# ...and what a config file legitimately holds under a credential-shaped key stays quiet.
IGNORE   token: ${{ secrets.GITHUB_TOKEN }}
IGNORE   GH_TOKEN: ${{secrets.GITHUB_TOKEN}}
IGNORE   POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:?set it in code/.env}
IGNORE   password_file: /run/secrets/db-password
IGNORE   token_ttl: 3600
IGNORE PASSWORD_FILE=/run/secrets/db-password some-command --flag
IGNORE   E2E_JWT_SECRET="$(head -c 32 /dev/urandom | base64 | tr -d '\n')"

# --- a public identifier and a mock are not credentials (added 2026-10-01) ----------------
# Found when testing/ was first scanned: Windows unattend.xml names each component by its
# PUBLIC strong-name token, and an E2E helper seeds a mock token into localStorage.
IGNORE                publicKeyToken="31bf3856ad364e35"
IGNORE <assemblyIdentity name="Microsoft-Windows-Setup" publicKeyToken="31bf3856ad364e35" language="neutral"/>
IGNORE         accessToken: 'mock_token_for_testing',
IGNORE   apiKey: 'fake-key-for-the-fixture-only'
# Exact names and prefixes only: a private key token, and a literal merely CONTAINING mock, stay findings.
CATCH const privateKeyToken = 'q7Xm2Kd9Lp4Rt1'
CATCH accessToken: 'Rt1mock_q7Xm2Kd9'

# --- EVERY assignment on a line is judged, not only the first (added 2026-10-01) ----------
# The scan read the line's first match and stopped when the validator rejected it, so an XML
# element whose first attribute is not credential-shaped hid every attribute after it.
CATCH <connection host="db.internal.example" password="q7Xm2Kd9Lp4Rt1"/>
CATCH $cfg = ['host' => 'db.internal.example', 'password' => 'q7Xm2Kd9Lp4Rt1'];

# --- found when the repo-root .github/ was first scanned (added 2026-10-01) ----------------
# A key that is the past tense of `pass` is not a password field.
IGNORE               'passed': ':white_check_mark: Passed',
CATCH DB_PASS=q7Xm2Kd9Lp4Rt1
# A sentence that names an identifier with an underscore in it is still a sentence.
IGNORE       "notACredential": "An hCaptcha SITE key, public by construction: the VITE_ prefix bakes it into the client bundle."
# A demo fixture's value is a placeholder of the same class as example and dummy.
IGNORE         privateKey: 'demo_private_AAAAAAAAAAAAAAAAAAAAAAAAAAAA',

# --- a key INSIDE a quoted string is not an assignment (2026-10-02) -----------------------------
# A handoff's grep pattern for a setting, as prose: the quote after the colon CLOSES the string the
# key began in. The quoted-value branch read it as a key and a value, and flagged a number of hours.
IGNORE      check: grep -n 'setPasswordLinkHours:' /srv/app/policy.ts → expect: 'the hours set there'
# ...and the exemption must reach neither a properly quoted key nor an assignment inside a string
CATCH $cfg = {'password': 'q7Xm2Kd9Lp4Rt1'}
CATCH docker run -e 'DB_PASSWORD="q7Xm2Kd9Lp4Rt1"' app
# A rejected match must not hide an assignment that begins inside it: `host` is not a secret name,
# and that match ran from the first quote to the third, past the real one.
CATCH echo 'host:' x; DB_PASSWORD='q7Xm2Kd9Lp4Rt1'
