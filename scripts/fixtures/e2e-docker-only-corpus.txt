# Regression corpus for audit:e2e-docker-only.
#
# The auditor refuses to scan unless every line here classifies correctly. Both halves matter and
# both are load-bearing: the CATCH half proves it can find a bare runner invocation at all, and the
# IGNORE half proves it holds its fire on the many places this repo legitimately WRITES the phrase
# "npx playwright test" — in comments forbidding it, in prose, and inside the Docker harness where
# the invocation is correct. A rule that flagged those would be switched off within a day, after
# which it would catch nothing.
#
# Format:  CATCH <line>   — must be reported as a violation
#          IGNORE <line>  — must NOT be reported
#          CATCH:json / IGNORE:json — the same, judged as a line of a JSON file
# Blank lines and lines starting with # are skipped.

# ── Must catch: a real bare invocation at a command position ─────────────────────────────────
CATCH npx playwright test
CATCH   npx playwright test --workers=4
CATCH npx playwright test testing/e2e/canvas.spec.ts
CATCH npm run lint && npx playwright test
CATCH ./node_modules/.bin/playwright test
CATCH npm exec playwright test
CATCH yarn playwright test
CATCH pnpm playwright test
CATCH npx playwright test --ui

# ── Must catch: the command inside a STRING, which is how it actually gets committed ──────────
# A negative test against a real repo file caught the auditor missing every one of these, because
# an earlier version stripped quoted strings to avoid firing on prose. Quoting is not evidence
# that something is not a command; printing is.
CATCH execSync("npx playwright test")
CATCH execSync('npx playwright test --workers=1')
CATCH   "test:e2e": "npx playwright test",
CATCH     run: npx playwright test
CATCH command: npx playwright test --reporter=list
CATCH const cmd = 'npx playwright test'

# ── Must ignore: an explicit, reasoned suppression (bare suppression cannot match) ────────────
IGNORE const forbidden = 'npx playwright test' // e2e-docker-only-ignore: the string this rule detects

# ── Must ignore: the invocation is inside Docker, which is the whole point ────────────────────
IGNORE docker compose run --rm playwright-tests npx playwright test
IGNORE docker compose --profile single up playwright-single --build
IGNORE docker exec quasar-e2e-iterate npx playwright test testing/e2e/canvas.spec.ts
IGNORE ./scripts/run-tests.sh
IGNORE ./scripts/run-foundry.sh testing/e2e/canvas.spec.ts

# ── Must ignore: comments, including ones that FORBID the thing ───────────────────────────────
IGNORE # E2E runs via Docker, never bare npx playwright test
IGNORE // Never `npx playwright test` — the browser must come from the image, not the machine
IGNORE   # npx playwright test would bypass the harness entirely
IGNORE * runs happen in the Docker image, not via npx playwright test

# ── Must ignore: the phrase inside a quoted string is prose, not a command ────────────────────
IGNORE echo "use the docker harness, not npx playwright test"
IGNORE echo 'npx playwright test is forbidden here'

# ── Must ignore: near-misses that are not the test runner ────────────────────────────────────
IGNORE npx playwright install-deps
IGNORE npx playwright install chromium
IGNORE npm install --save-dev @playwright/test@1.58.2
IGNORE import { test, expect } from '@playwright/test'
IGNORE const { chromium } = require('playwright')

# ── Must ignore: a SEARCH for the string is looking for the command, not running it ──────────
# A handoff's check that a doc no longer teaches the bare runner, as it reaches the generated
# handoff corpus (Gantry, 2026-10-02). Only the search tool's quoted pattern is set aside.
IGNORE grep -c 'npx playwright test' docs/PLAYWRIGHT_LOCAL_TESTING.md
IGNORE       "command": "grep -c 'npx playwright test' /srv/repo/docs/E2E.md",
IGNORE rg -n "npx playwright test" docs/
# ...and a real run on the same line as a search is still a run
CATCH grep -q ready logs/app.log && npx playwright test
CATCH grep -c 'npx playwright test' docs/E2E.md; npx playwright test

# ── Must catch: printing or docker in ONE command does not exempt the next ───────────────────
# PRINTS and VIA_DOCKER exempted the whole LINE (found 2026-10-02 while adding the search case), so
# a script that announced its run, or touched docker first, ran the bare runner unseen.
CATCH echo "running e2e"; npx playwright test
CATCH docker ps -q && npx playwright test --workers=2
# ...while a printed or containerised runner stays exempt, quotes and all
IGNORE echo "step 2; then npx playwright test runs in the image"
IGNORE docker compose run --rm tests sh -c "cd /app && npx playwright test"

# ── A backtick means different things by language ──────────────────────────────────────────────
# In TypeScript a template literal handed to a shell is a command, and in shell a backtick is a
# command substitution. In JSON a backtick is only markdown inline code inside a string: a knowledge
# entry's title projected into a generated export was reported as a runner on 2026-10-02. An npm
# script, the one place a JSON string is shell, is still anchored by its own quote.
CATCH execSync(`npx playwright test`)
CATCH result=`npx playwright test --reporter=list`
CATCH:json   "test:e2e": "npx playwright test",
CATCH:json   "e2e": "npm run build && npx playwright test --workers=1",
IGNORE:json        "title": "Under `set -e`, `npx playwright test; RC=$?` ends the script at the first failing run",
IGNORE:json        "body": "**Problem:** E2E entrypoints ran the suite as `npx playwright test` and read `$?` next",
